
// --- API ENDPOINTS ---

/**
 * @api {get} /api/rates Request currency rates for a date range.
 * @apiName GetRates
 * @apiGroup Rates
 *
 * @apiParam {String} startDate Start date in YYYY-MM-DD format.
 * @apiParam {String} endDate End date in YYYY-MM-DD format.
 * @apiParam {String} currencies Comma-separated list of currency codes (e.g., 'usd,eur,rub').
 *
 * @apiSuccess {Object} data Object with currency codes as keys and arrays of rates as values.
 *
 * @apiSuccessExample {json} Success-Response:
 *     HTTP/1.1 200 OK
 *     {
 *       "usd": [
 *         { "date": "2024-01-01", "rate": "16.1" },
 *         { "date": "2024-01-02", "rate": "16.1" }
 *       ],
 *       "eur": [
 *         { "date": "2024-01-01", "rate": "18.0401" },
 *         { "date": "2024-01-02", "rate": "18.0512" }
 *       ]
 *     }
 *
 * @apiError (400 Bad Request) {String} error Missing or invalid parameters.
 */
app.get('/api/rates', async (req: Request, res: Response) => {
  const { startDate, endDate, currencies } = req.query;

  // 1. Input Validation
  if (
    !startDate ||
    !endDate ||
    !currencies ||
    typeof startDate !== 'string' ||
    typeof endDate !== 'string' ||
    typeof currencies !== 'string'
  ) {
    return res.status(400).json({
      error:
        'Missing or invalid parameters. Please provide startDate, endDate, and currencies.',
    });
  }

  const dateRegex = /^\d{4}-\d{2}-\d{2}$/;
  if (!dateRegex.test(startDate) || !dateRegex.test(endDate)) {
    return res.status(400).json({
      error: 'Invalid date format. Please use YYYY-MM-DD.',
    });
  }

  if (startDate > endDate) {
    return res
      .status(400)
      .json({ error: 'startDate cannot be after endDate.' });
  }

  const currencyCodes = currencies
    .toLowerCase()
    .split(',')
    .filter((c) => c.trim() !== '');
  if (currencyCodes.length === 0) {
    return res
      .status(400)
      .json({ error: 'The currencies parameter cannot be empty.' });
  }

  // 2. Fetch and Process Data
  try {
    const data = await getRatesForCurrencies(currencyCodes, startDate, endDate);
    return res.status(200).json(data);
  } catch (error) {
    console.error('Error fetching currency data:', error);
    return res
      .status(500)
      .json({ error: 'An internal server error occurred.' });
  }
});

/**
 * @api {get} /api/rates/:currency/latest Request the latest available rate for a single currency.
 * @apiName GetLatestRate
 * @apiGroup Rates
 *
 * @apiParam {String} currency The three-letter currency code (e.g., 'usd', 'eur').
 *
 * @apiSuccess {Object} data The latest rate record containing the date and rate.
 *
 * @apiSuccessExample {json} Success-Response (USD):
 *     HTTP/1.1 200 OK
 *     {
 *       "date": "2025-10-18",
 *       "rate": "16.1000"
 *     }
 *
 * @apiError (404 Not Found) {String} error The currency data was not found or the file was empty.
 */
app.get('/api/rates/:currency/latest', async (req: Request, res: Response) => {
  const currencyCode = req.params.currency.toLowerCase();
  const filePath = path.join(DATA_DIR, `${currencyCode}.csv`);

  // 1. Check if the data file exists
  try {
    await access(filePath);
  } catch {
    return res
      .status(404)
      .json({ error: `Data for currency '${currencyCode}' not found.` });
  }

  // 2. Fetch the latest record
  try {
    const latestRecord = await getLatestRate(filePath);

    if (latestRecord) {
      return res.status(200).json(latestRecord);
    }
    // This case handles a file that exists but is empty (or contains only a header)
    return res.status(404).json({
      error: `No rate records found for currency '${currencyCode}'.`,
    });
  } catch (error) {
    console.error(`Error fetching latest rate for ${currencyCode}:`, error);
    return res
      .status(500)
      .json({ error: 'An internal server error occurred.' });
  }
});

app.get('/', (_req: Request, res: Response) => {
  res.send('Currency Rate API is running. Try /api/rates');
});
